Smart Contract: FlashBets.sol
solidity
// contracts/FlashBets.sol
contract FlashBets is ReentrancyGuard, Ownable {
    
    enum Category { SPORTS, POLITICS, ENTERTAINMENT }
    enum Status { PENDING, ACTIVE, CLOSED, RESOLVED }
    
    struct FlashMarket {
        uint256 id;
        string question;
        Category category;
        uint256 startTime;      // When betting opens
        uint256 closeTime;      // startTime + 15 mins
        uint256 resolveTime;    // When result is known
        Status status;
        bool outcome;           // true = YES won, false = NO won
        uint256 yesPool;
        uint256 noPool;
    }
    
    struct FlashBet {
        address user;
        uint256 marketId;
        bool predictedYes;
        uint256 amount;
        bool claimed;
    }
    
    // Constants
    uint256 public constant FLASH_DURATION = 15 minutes;
    uint256 public constant FLASHES_PER_DAY = 2;
    uint256 public constant FLASH_TIMES = [12 hours, 20 hours]; // 12 PM, 8 PM UTC
    
    mapping(uint256 => FlashMarket) public flashMarkets;
    mapping(uint256 => mapping(address => FlashBet)) public flashBets;
    uint256 public flashMarketCounter;
    
    // Backend automation address (can create markets)
    address public automation;
    
    event FlashMarketCreated(uint256 indexed marketId, string question, Category category);
    event FlashBetPlaced(uint256 indexed marketId, address indexed user, bool predictedYes, uint256 amount);
    event FlashMarketResolved(uint256 indexed marketId, bool outcome);
    
    modifier onlyAutomation() {
        require(msg.sender == automation, "Only automation");
        _;
    }
    
    /**
     * @notice Backend calls this at 11:55 AM and 7:55 PM daily
     */
    function createFlashMarket(
        string memory question,
        Category category,
        uint256 resolveTime
    ) external onlyAutomation {
        uint256 marketId = flashMarketCounter++;
        
        FlashMarket storage market = flashMarkets[marketId];
        market.id = marketId;
        market.question = question;
        market.category = category;
        market.startTime = block.timestamp;
        market.closeTime = block.timestamp + FLASH_DURATION;
        market.resolveTime = resolveTime;
        market.status = Status.ACTIVE;
        
        emit FlashMarketCreated(marketId, question, category);
    }
    
    /**
     * @notice Users place bets during 15-min window
     */
    function placeFlashBet(
        uint256 marketId,
        bool predictYes,
        uint256 amount
    ) external nonReentrant {
        FlashMarket storage market = flashMarkets[marketId];
        
        require(market.status == Status.ACTIVE, "Market not active");
        require(block.timestamp < market.closeTime, "Betting closed");
        require(amount > 0, "Invalid amount");
        
        // Transfer USDC from user
        usdc.transferFrom(msg.sender, address(this), amount);
        
        // Record bet
        FlashBet storage bet = flashBets[marketId][msg.sender];
        require(bet.amount == 0, "Already bet on this flash");
        
        bet.user = msg.sender;
        bet.marketId = marketId;
        bet.predictedYes = predictYes;
        bet.amount = amount;
        
        // Update pools
        if (predictYes) {
            market.yesPool += amount;
        } else {
            market.noPool += amount;
        }
        
        emit FlashBetPlaced(marketId, msg.sender, predictYes, amount);
    }
    
    /**
     * @notice Backend resolves after event concludes
     */
    function resolveFlashMarket(
        uint256 marketId,
        bool outcome
    ) external onlyAutomation {
        FlashMarket storage market = flashMarkets[marketId];
        
        require(market.status == Status.ACTIVE, "Not active");
        require(block.timestamp >= market.resolveTime, "Too early");
        
        market.status = Status.RESOLVED;
        market.outcome = outcome;
        
        emit FlashMarketResolved(marketId, outcome);
    }
    
    /**
     * @notice Winners claim payouts
     */
    function claimWinnings(uint256 marketId) external nonReentrant {
        FlashMarket storage market = flashMarkets[marketId];
        FlashBet storage bet = flashBets[marketId][msg.sender];
        
        require(market.status == Status.RESOLVED, "Not resolved");
        require(bet.amount > 0, "No bet placed");
        require(!bet.claimed, "Already claimed");
        require(bet.predictedYes == market.outcome, "You lost");
        
        bet.claimed = true;
        
        // Calculate payout: (userBet / winningPool) * totalPool
        uint256 winningPool = market.outcome ? market.yesPool : market.noPool;
        uint256 totalPool = market.yesPool + market.noPool;
        
        // Take 5% platform fee
        uint256 feeAmount = (totalPool * 5) / 100;
        uint256 payoutPool = totalPool - feeAmount;
        
        uint256 payout = (bet.amount * payoutPool) / winningPool;
        
        usdc.transfer(msg.sender, payout);
        
        // Award TRENCHY points (2x for FlashBets)
        awardPoints(msg.sender, payout * 20); // 20 points per $1
    }
}
